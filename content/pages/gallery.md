<script>
document.addEventListener('DOMContentLoaded', function() {
    let currentVessel = 'all';
    let currentYear = 'all';

    const vesselPills = document.querySelectorAll('.gallery-filter-pill[data-vessel]');
    const yearPills = document.querySelectorAll('.gallery-filter-pill[data-year]');
    const galleryItems = document.querySelectorAll('.gallery-card-item');
    const statusTitle = document.getElementById('gallery-status-title');
    const noResults = document.getElementById('no-gallery-results');

    function filterGallery() {
        let visibleCount = 0;

        galleryItems.forEach(item => {
            const itemVessel = item.getAttribute('data-vessel');
            const itemYear = item.getAttribute('data-year');

            const matchVessel = (currentVessel === 'all' || itemVessel === currentVessel);
            const matchYear = (currentYear === 'all' || itemYear === currentYear);

            if (matchVessel && matchYear) {
                item.style.display = 'block';
                visibleCount++;
            } else {
                item.style.display = 'none';
            }
        });

        let vesselText = currentVessel === 'all' ? 'All Fleet' : (currentVessel === '77m' ? '77M Workboat' : (currentVessel === 'hibernia' ? 'Barge Hibernia' : 'Little Knocknagow'));
        let yearText = currentYear === 'all' ? 'All Eras' : currentYear;

        statusTitle.textContent = `Displaying ${vesselText} Photos (${yearText}) — ${visibleCount} Photos`;

        if (visibleCount === 0) {
            noResults.style.display = 'block';
        } else {
            noResults.style.display = 'none';
        }
    }

    vesselPills.forEach(pill => {
        pill.addEventListener('click', function() {
            vesselPills.forEach(p => p.classList.remove('active'));
            this.classList.add('active');
            currentVessel = this.getAttribute('data-vessel');
            filterGallery();
        });
    });

    yearPills.forEach(pill => {
        pill.addEventListener('click', function() {
            yearPills.forEach(p => p.classList.remove('active'));
            this.classList.add('active');
            currentYear = this.getAttribute('data-year');
            filterGallery();
        });
    });
});
</script>
